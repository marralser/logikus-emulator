cmake_minimum_required(VERSION 3.16)
project(logikus_linux)
set(CMAKE_CXX_STANDARD 17)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL2 REQUIRED sdl2)
add_executable(logikus src/main.cpp)
target_include_directories(logikus PRIVATE ${SDL2_INCLUDE_DIRS})
target_compile_options(logikus PRIVATE ${SDL2_CFLAGS_OTHER})
target_link_libraries(logikus PRIVATE ${SDL2_LIBRARIES})
